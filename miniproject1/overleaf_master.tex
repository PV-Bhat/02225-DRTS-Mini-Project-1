\section{Introduction}
This version syncs the harmonic evaluation pipeline and jitter-aware DM analysis.

\section{Methodology}
\subsection{Deadline Monotonic Analysis (RTA)}
We use jitter-aware fixed-point RTA:
\[
R_i^{(k+1)} = C_i + \sum_{j\in hp(i)}\left\lceil\frac{R_i^{(k)}+J_j}{T_j}\right\rceil C_j.
\]

\section{Implementation}
The implementation is Python-based and split into modules: `Task-model.py` for CSV parsing/data model, `DM-logic.py` for fixed-priority RTA, `EDF-logic.py` for PDC and exact EDF WCRT simulation, `simulation.py` for stochastic Monte Carlo, and `analysis.py` for orchestration.

\subsection*{RTA Loop (with Jitter)}
\begin{algorithm}[H]
\caption{DM RTA with release jitter}
\begin{algorithmic}[1]
\For{task $\tau_i$ in DM-priority order}
\State $R \gets C_i$
\While{true}
\State $I \gets \sum_{j\in hp(i)} \lceil (R + J_j)/T_j \rceil C_j$
\State $R' \gets C_i + I$
\If{$R' = R$} \State \textbf{break} \EndIf
\State $R \gets R'$
\If{$R > D_i$} \State \textbf{break} \EndIf
\EndWhile
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection*{Event-driven EDF Min-Heap Simulator}
\begin{algorithm}[H]
\caption{Exact EDF simulation over $[0,H)$}
\begin{algorithmic}[1]
\State Generate all jobs with $(r_{i,k}, d_{i,k}, C_i)$ for $kT_i<H$
\State Push released jobs into min-heap keyed by absolute deadline
\While{jobs remain or heap non-empty}
\State Advance to next release/completion event
\State Preempt if a new earliest deadline arrives
\State On completion, record $R_{i,k}=f_{i,k}-r_{i,k}$
\EndWhile
\State $WCRT_i = \max_k R_{i,k}$
\end{algorithmic}
\end{algorithm}

\subsection{Evaluation with Harmonic Task Sets}
A parallel dataset was generated using a semi-harmonic period pool:
$[1000, 2000, 4000, 5000, 10000, 20000, 40000, 50000, 100000]$.
This keeps hyperperiods tractable for exact EDF simulation. On 80 generated sets,
DM passes 76/80 and EDF passes 79/80; the single EDF failure appears at 90\% utilization and is consistent with constrained-deadline demand behavior in this synthetic set family.
